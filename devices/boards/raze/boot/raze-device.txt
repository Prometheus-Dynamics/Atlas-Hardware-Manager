# Raze device configuration. The OS config.txt includes this file at top level.
# Only the product's own overlays are here; the board's device tree comes from one generated overlay
# (boards/raze/overlay/README.txt, pending Lemnos). Each line has its reason.

# GPU (V3D) and display (KMS). The product's choice: HeliOS wants it, and dma-buf frame streams
# allocate from it.
dtoverlay=vc4-kms-v3d-pi5

# Kernel logs kept in RAM across a reset (pstore): 256 KiB at 0x0b000000, 128 KiB of it the console
# log and two 64 KiB panic/oops records. Kept for the support bundle: a crash on a robot leaves its
# last log. Sizes are dev's, measured on a Raze.
dtoverlay=ramoops-pi5,total-size=0x40000,record-size=0x10000,console-size=0x20000

# External USB-A (GPIO20) and USB-C (GPIO16) port power: driven high by the firmware before Linux, so
# the ports are never off at boot. Claims nothing; Lemnos's gpio-power-switch then takes the lines.
gpio=16,20=op,dh

[all]

# The bcm2835 hardware watchdog, on from the start. systemd arms it (RuntimeWatchdogSec=15s,
# raze-base-files), so a hung boot resets. Without this line systemd has no watchdog device.
dtparam=watchdog=on
